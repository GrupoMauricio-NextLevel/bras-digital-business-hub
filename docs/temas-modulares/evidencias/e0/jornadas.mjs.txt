import { chromium } from "playwright-core";
const B = "http://localhost:3100", OUT = "docs/temas-modulares/evidencias/e0";
const b = await chromium.launch({ executablePath: "/opt/pw-browsers/chromium" });
const log = (...a) => console.log(...a);
// 1. Visitante: galeria, detalhe, demo (Commerce)
{
  const ctx = await b.newContext({ viewport: { width: 1440, height: 900 }, extraHTTPHeaders: { "x-forwarded-host": "lojas.nextlevelgroup.com.br" } });
  const p = await ctx.newPage();
  await p.goto(B + "/temas"); await p.waitForTimeout(800); await p.screenshot({ path: `${OUT}/galeria-temas-1440.png`, fullPage: true });
  log("galeria: temas listados =", await p.$$eval("a[href^='/temas/']", (as) => [...new Set(as.map((a) => a.getAttribute("href")))].join(", ")));
  log("galeria: tem filtro de nível/nicho?", await p.locator("select, [role=tablist], [role=radiogroup]").count());
  await p.goto(B + "/temas/lume"); await p.waitForTimeout(600); await p.screenshot({ path: `${OUT}/detalhe-lume-1440.png`, fullPage: true });
  await p.goto(B + "/store/demo-lume"); await p.waitForTimeout(800); await p.screenshot({ path: `${OUT}/demo-lume-home-1440.png` });
  // tenta comprar na demo
  const prod = await p.$eval("a[href*='/product/']", (a) => a.getAttribute("href")).catch(() => null);
  log("demo: primeiro produto =", prod);
  if (prod) { await p.goto(B + prod); await p.waitForTimeout(600); const btn = p.getByRole("button", { name: /Adicionar|Comprar/i }).first(); log("demo: botão de compra visível =", await btn.isVisible().catch(() => false));
    if (await btn.isVisible().catch(() => false)) { await btn.click().catch(() => {}); await p.waitForTimeout(800); }
    await p.goto(B + "/store/demo-lume/cart"); await p.waitForTimeout(700); await p.screenshot({ path: `${OUT}/demo-lume-carrinho-1440.png`, fullPage: true });
    log("demo: texto do carrinho =", (await p.locator("main").innerText()).replace(/\s+/g, " ").slice(0, 300)); }
  await p.goto(B + "/"); await p.waitForTimeout(500);
  log("home commerce: textos de preço =", (await p.locator("#valores").innerText().catch(() => "")).replace(/\s+/g, " ").slice(0, 400));
  await ctx.close();
}
// 2. Lojista SuperDrop (STORE_SUBDOMAIN) no editor
{
  const ctx = await b.newContext({ viewport: { width: 1440, height: 900 }, extraHTTPHeaders: { "x-forwarded-host": "superdrop.com.br" } });
  const p = await ctx.newPage(); p.setDefaultTimeout(60000);
  await p.goto(B + "/login"); await p.fill("input[type=email]", "loja@mrssports.demo"); await p.fill("input[type=password]", "demo1234"); await p.keyboard.press("Enter"); await p.waitForURL((u) => !u.pathname.startsWith("/login"));
  await p.goto(B + "/theme-editor"); await p.waitForTimeout(3000); await p.screenshot({ path: `${OUT}/editor-superdrop-subdominio-1440.png` });
  log("editor subdomínio: aviso =", (await p.locator("aside").first().innerText()).replace(/\s+/g, " ").slice(0, 220));
  await p.goto(B + "/merchant/themes").catch(() => {}); await p.waitForTimeout(800); await p.screenshot({ path: `${OUT}/lojista-temas-superdrop-1440.png`, fullPage: true });
  log("lojista temas url =", p.url());
  await ctx.close();
}
await b.close();
